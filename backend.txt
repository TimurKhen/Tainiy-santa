async def create_exchange(initData):
    headers = {"Authorization": f"tma {initData}"}
    form = aiohttp.FormData()
    form.add_field("name", "ФЫВА")
    form.add_field("place", "ЙЦУКЕН")
    form.add_field("money", "100")
    form.add_field("distribution_date", "2026-11-18T00:00:00.000Z")
    form.add_field("swap_date", "2026-11-18T20:00:00.000Z")
    form.add_field("icon", "burda")


    async with aiohttp.ClientSession(headers=headers) as session:
        async with session.post("http://localhost:4444/api/groups/create", data=form) as resp:
            body = await resp.text()
            print(f"HTTP {resp.status}")
            print(body)

asyncio.run(create_exchange(initDataJ))
